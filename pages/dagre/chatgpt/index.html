const g = new dagre.graphlib.Graph({ multigraph: true }); g.setGraph({ rankdir: 'LR', ranker: 'network-simplex', nodesep: 45, ranksep: 70 }); g.setDefaultEdgeLabel(() => ({})); g.setNode('A', { width: 140, height: 54 }); g.setNode('B', { width: 140, height: 54 }); g.setEdge('A', 'B', { minlen: 2, weight: 3 }); dagre.layout(g); console.log(g.node('A')); // x, y console.log(g.edge('A','B')); // points[]
