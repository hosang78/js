회원 등록

Prepared statement + named bind

const stmt = db.prepare(`
 INSERT INTO members(name, grade, branch_id, active)
 VALUES ($name, $grade, $branch, $active)
`);
stmt.run({$name, $grade, $branch, $active});
stmt.free();
