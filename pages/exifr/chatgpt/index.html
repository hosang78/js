사용 중인 핵심 API

BlobReader/Writer, TextReader/Writer, Uint8ArrayReader/Writer, ZipReader, ZipWriter, Entry.getData(), onprogress, AbortSignal, password, encryptionStrength, zipCrypto, level, compressionMethod, directory, comment, useWebWorkers, useCompressionStream, zip64.

// ZIP 생성
const writer = new zip.ZipWriter(new zip.BlobWriter("application/zip"), {
  level: 6,
  password: "secret",
  encryptionStrength: 3
});
await writer.add("hello.txt", new zip.TextReader("안녕하세요"), {
  onprogress(index, max) { /* 진행률 */ }
});
const blob = await writer.close("archive comment");

// ZIP 읽기와 해제
const reader = new zip.ZipReader(new zip.BlobReader(blob), { password: "secret" });
const entries = await reader.getEntries();
const text = await entries[0].getData(new zip.TextWriter());
await reader.close();
