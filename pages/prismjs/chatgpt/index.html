async function loadTasks(endpoint) {
  const response = await fetch(endpoint, { headers: { Accept: 'application/json' } });
  if (!response.ok) throw new Error(`HTTP ${response.status}`);
  return response.json();
}

loadTasks('/api/tasks').then(console.table).catch(console.error);

3. 줄 강조 및 시작 줄 번호

data-line, data-start 속성으로 특정 범위와 실제 문서 줄 번호 표현.

from dataclasses import dataclass

@dataclass
class Vessel:
    name: str
    capacity: int

def available(vessels):
    return [v for v in vessels if v.capacity > 0]

print(available([Vessel("Blue", 12)]))

6. 괄호 매칭

코드의 괄호 위에 마우스를 올리거나 클릭해 짝을 확인.

function summarize(items) {
  return items.map((item) => ({
    ...item,
    score: Math.max(0, Math.min(100, item.score ?? 0))
  })).filter(({ score }) => score >= 70);
}

Java, SQL, JSON, YAML, Markdown 등 컴포넌트별 로딩.

WITH ranked AS (
  SELECT department, task_name, hours_saved,
         ROW_NUMBER() OVER (PARTITION BY department ORDER BY hours_saved DESC) AS rn
  FROM automation_tasks
)
SELECT * FROM ranked WHERE rn <= 3;
service:
  name: automation-demo
  port: 8875
  database: sqlite

  <script>document.querySelector('.notice').dataset.ready = 'true';</script>
</article>

14. Keep Markup

코드 내부의 기존 HTML 마크업을 유지하면서 강조.

  <mark>중요</mark>
  <span>처리 완료</span>
</div>
